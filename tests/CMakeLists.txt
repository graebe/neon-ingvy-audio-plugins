# THE CURVE ORACLE, both halves.
#
# The editor draws the envelope, so shape() exists TWICE -- once in the engine
# and once in JavaScript -- and the JS copy had the S-curve's first half
# un-mirrored for as long as it existed. Nothing caught it, because a wrong
# curve is still a curve.
#
# So both sides are pinned to one table generated BY THE ENGINE, and neither is
# compared against the other:
#
#   tg_curves     the engine still agrees with ui/test/shape_table.txt
#   tg_curves_js  ui/src/lib/curves.js still agrees with the same file
#
# Regenerate the fixture with `tg_shape_table > ui/test/shape_table.txt`, and
# only when the curve maths is MEANT to change.

    ${CMAKE_SOURCE_DIR}/plugins/trance-gate/ui/test/shape_table.txt)

endif()

# THE ENVELOPE ORACLE, and it is MEASURED rather than transcribed.
#
# The curve table above pins shape(); this pins the whole stage machine around
# it -- when the gate shuts, what the release falls from, what a zero-length
# stage does, and what the attack starts from when the previous release is
# still sounding. tg_envelope_table runs the REAL engine with a DC input at
# amount 1, where the gain it applies IS the envelope, so the reference owes
# nothing to anyone's reading of the Rust.
#
# It found two rules the UI had wrong and a third nobody had modelled at all.
set(TG_ENVELOPE_TABLE
    ${CMAKE_SOURCE_DIR}/plugins/trance-gate/ui/test/envelope_table.txt)

add_executable(tg_envelope_table envelope_table.c)
target_link_libraries(tg_envelope_table PRIVATE tg_engine)
if (NOT WIN32)
    target_link_libraries(tg_envelope_table PRIVATE m)
endif()
add_test(NAME tg_envelope COMMAND tg_envelope_table --verify ${TG_ENVELOPE_TABLE})

# THE FADE ORACLE, measured the same way the envelope one is.
#
# ui/src/lib/fade.js is the fade's formula in JavaScript, because three views
# need it -- the pads, the ring and the pattern plot -- and pushing a weight
# table thirty times a second to save three lines of arithmetic is the wrong
# trade. A second implementation of the DSP is only tolerable pinned, so it is.
#
# The reference is the engine's own GAIN: DC in, no envelope, so the gain during
# a step IS that step's arrival weight. If the weight table and the gain law ever
# disagree, this reports the gain law -- the one a listener hears.

    ${CMAKE_SOURCE_DIR}/plugins/trance-gate/ui/test/fade_table.txt)

# The installed AU, rendered by a host that actually supplies a transport.
# AVAudioEngine supplies none, so a plugin under it correctly holds the gate
# open and renders dry -- a green render that proves nothing. This one sets
# the host callbacks the AU wrapper reads, drives the macros through the AU
# parameter API, and compares against the engine directly.
#
# It finds the plugin by the (aufx, TrGt, Grbe) triple, which the iPlug2 build
# carries over from the JUCE one -- so this test covers the new AU without a line
# changing in it.
#
# WHICH IS ALSO ITS ONE WEAKNESS, AND WHY THE VERSION IS PASSED IN. Those three
# values are the plugin's identity and never move, so EVERY build it has ever had
# claims them -- including one left installed under an older bundle name. The
# rename to NITranceGate left TranceGate.component in place, macOS registered the
# old one, and this test went on passing against a binary with none of the new
# work in it. PLUG_VERSION_HEX is read out of config.h here and handed to the test
# so it can refuse a bundle that is not the one this tree builds.
if (APPLE)
    file(READ ${CMAKE_SOURCE_DIR}/plugins/trance-gate/config.h TG_CONFIG_H)
    string(REGEX MATCH "#define[ \t]+PLUG_VERSION_HEX[ \t]+(0x[0-9a-fA-F]+)"
           _tg_ver_match "${TG_CONFIG_H}")
    set(TG_PLUG_VERSION_HEX "${CMAKE_MATCH_1}")
    if (NOT TG_PLUG_VERSION_HEX)
        message(FATAL_ERROR
            "No PLUG_VERSION_HEX in plugins/trance-gate/config.h -- tg_au cannot "
            "tell the bundle it built from one shadowing it")
    endif()

    add_executable(tg_au_render au_render.c)
    target_compile_definitions(tg_au_render
        PRIVATE EXPECTED_AU_VERSION=${TG_PLUG_VERSION_HEX})
    target_link_libraries(tg_au_render PRIVATE tg_engine m
        "-framework AudioToolbox" "-framework AudioUnit" "-framework CoreFoundation")
    add_test(NAME tg_au COMMAND tg_au_render)
endif()

# THE ONE THAT JUSTIFIES SHARED MEMORY.
#
# Every other test here would pass over a process-local ring. This one forks,
# writes in the child and reads in the parent -- which is the thing a static
# table inside the library could never do, and the reason the transport is
# shm_open rather than a `static BUSES[16]`. Without it, the day a host
# sandboxes a plugin the failure is a receiver that is simply always empty.
if (NOT WIN32)
    add_executable(abus_ipc abus_ipc.c)
    target_link_libraries(abus_ipc PRIVATE audio_bus_engine)
    add_test(NAME abus_ipc COMMAND abus_ipc)
    # It waits on a forked child through usleep loops; the default 1500s is
    # generous but a hang here should be a failure, not a stuck suite.
    set_tests_properties(abus_ipc PROPERTIES TIMEOUT 60)
endif()
