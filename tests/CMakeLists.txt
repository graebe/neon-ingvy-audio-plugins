# The engine's own tests, built against the SUBMODULE -- the same library the
# plugin links. If these pass here and fail on Move, the submodule pin moved.
#
# tg_engine is the submodule's Rust crate, built once at the top level; see
# cmake/TgEngine.cmake.
#
# The C targets were three after JUCE went; tg_params, tg_vst3 and tg_shot hosted JUCE
# classes directly -- they instantiated the AudioProcessor, hosted the VST3
# through JUCE's plugin host, and rendered the editor -- so they could not
# outlive it. What remains is everything that was written in plain C against
# the engine's ABI, which is also everything that checks the SOUND.
set(TG_ENGINE ${CMAKE_SOURCE_DIR}/engines/trance-gate)

add_executable(tg_core_tests ${TG_ENGINE}/tests/test_core.c)
target_link_libraries(tg_core_tests PRIVATE tg_engine)
if (NOT WIN32)
    target_link_libraries(tg_core_tests PRIVATE m)
endif()
add_test(NAME tg_core COMMAND tg_core_tests)

# Renders through the PLUGIN's audio path (float, split channels, a DAW-shaped
# transport) and compares against the Move module's reference render. This is
# the check that the port did not change the sound, and it has survived the
# engine going to Rust and the shell going from JUCE to iPlug2 unchanged --
# which is the point of it being plain C against the ABI.
add_executable(tg_render_plugin render_plugin.c)
target_link_libraries(tg_render_plugin PRIVATE tg_engine)
if (NOT WIN32)
    target_link_libraries(tg_render_plugin PRIVATE m)
endif()
add_test(NAME tg_render_ab COMMAND tg_render_plugin --verify)

# THE CURVE ORACLE, both halves.
#
# The editor draws the envelope, so shape() exists TWICE -- once in the engine
# and once in JavaScript -- and the JS copy had the S-curve's first half
# un-mirrored for as long as it existed. Nothing caught it, because a wrong
# curve is still a curve.
#
# So both sides are pinned to one table generated BY THE ENGINE, and neither is
# compared against the other:
#
#   tg_curves     the engine still agrees with ui/test/shape_table.txt
#   tg_curves_js  ui/src/lib/curves.js still agrees with the same file
#
# Regenerate the fixture with `tg_shape_table > ui/test/shape_table.txt`, and
# only when the curve maths is MEANT to change.
set(TG_SHAPE_TABLE
    ${CMAKE_SOURCE_DIR}/plugins/trance-gate/ui/test/shape_table.txt)

add_executable(tg_shape_table shape_table.c)
target_link_libraries(tg_shape_table PRIVATE tg_engine)
if (NOT WIN32)
    target_link_libraries(tg_shape_table PRIVATE m)
endif()
add_test(NAME tg_curves COMMAND tg_shape_table --verify ${TG_SHAPE_TABLE})

# THE ENVELOPE ORACLE, and it is MEASURED rather than transcribed.
#
# The curve table above pins shape(); this pins the whole stage machine around
# it -- when the gate shuts, what the release falls from, what a zero-length
# stage does, and what the attack starts from when the previous release is
# still sounding. tg_envelope_table runs the REAL engine with a DC input at
# amount 1, where the gain it applies IS the envelope, so the reference owes
# nothing to anyone's reading of the Rust.
#
# It found two rules the UI had wrong and a third nobody had modelled at all.
set(TG_ENVELOPE_TABLE
    ${CMAKE_SOURCE_DIR}/plugins/trance-gate/ui/test/envelope_table.txt)

add_executable(tg_envelope_table envelope_table.c)
target_link_libraries(tg_envelope_table PRIVATE tg_engine)
if (NOT WIN32)
    target_link_libraries(tg_envelope_table PRIVATE m)
endif()
add_test(NAME tg_envelope COMMAND tg_envelope_table --verify ${TG_ENVELOPE_TABLE})

# node's OWN test runner (22+), so the UI keeps a dependency tree of vite and
# solid and nothing else -- which is the point of the licence audit, not a
# preference. Skipped rather than failed when node is absent: a C++ developer
# building the plugin should not need a JS toolchain to run the DSP tests.
find_program(TG_NODE node)
if (TG_NODE)
    # The FILES, not the directory: `node --test <dir>` tries to execute the
    # directory as a module and dies with MODULE_NOT_FOUND.
    set(TG_UI_TEST ${CMAKE_SOURCE_DIR}/plugins/trance-gate/ui/test)
    add_test(NAME tg_curves_js
             COMMAND ${TG_NODE} --test ${TG_UI_TEST}/curves.test.mjs)
    set_tests_properties(tg_curves_js PROPERTIES
        ENVIRONMENT "TG_SHAPE_TABLE=${TG_SHAPE_TABLE}")

    add_test(NAME tg_envelope_js
             COMMAND ${TG_NODE} --test ${TG_UI_TEST}/envelope.test.mjs)
    set_tests_properties(tg_envelope_js PROPERTIES
        ENVIRONMENT "TG_ENVELOPE_TABLE=${TG_ENVELOPE_TABLE}")

    # THE TOKEN GUARD, which the JUCE build had and the web port lost. Uv.h
    # said "a ctest target fails the build if a colour literal appears in any
    # other source file"; four had crept into the UI by the time this was
    # written, one of them an arc glow at alpha 0.75 against the system's 0.45,
    # which no eye was ever going to catch.
    # ONE VERSION PER PRODUCT, spelled once. A product's version appears in
    # three or four files in three languages, nothing held them together, and
    # they had already come apart: the Spectrogram shipped PLUG_VERSION_STR
    # "1.0.0" against crates that said 0.1.0.
    add_test(NAME versions
             COMMAND ${TG_NODE} --test ${CMAKE_SOURCE_DIR}/tests/versions.test.mjs)

    # ui_tokens, not tg_tokens: it guards the WHOLE ui now -- the kit and both
    # editors -- where there used to be one per plugin, each blind to the other.
    add_test(NAME ui_tokens
             COMMAND ${TG_NODE} --test
                     ${CMAKE_SOURCE_DIR}/ui-kit/test/tokens.test.mjs)
else()
    message(STATUS "node not found -- skipping the UI's tests (curves, envelope, tokens)")
endif()

# The installed AU, rendered by a host that actually supplies a transport.
# AVAudioEngine supplies none, so a plugin under it correctly holds the gate
# open and renders dry -- a green render that proves nothing. This one sets
# the host callbacks the AU wrapper reads, drives the macros through the AU
# parameter API, and compares against the engine directly.
#
# It finds the plugin by the (aufx, TrGt, Grbe) triple, which the iPlug2 build
# carries over from the JUCE one -- so this test now covers the new AU without
# a line changing in it.
if (APPLE)
    add_executable(tg_au_render au_render.c)
    target_link_libraries(tg_au_render PRIVATE tg_engine m
        "-framework AudioToolbox" "-framework AudioUnit" "-framework CoreFoundation")
    add_test(NAME tg_au COMMAND tg_au_render)
endif()

# THE ANALYZER'S OWN TESTS, which are cargo's: the FFT against a naive DFT, the
# band mapping, and a counting global allocator proving the audio path allocates
# nothing. Run from here rather than left to be remembered.
#
# SPECTRO_ENV and SPECTRO_CARGO come from cmake/SpectroEngine.cmake, which found
# the toolchain the hard way (rustup's shims are often not on PATH).
# -p, not --manifest-path: there is ONE workspace and it is at the repository
# root, so the crates are selected by name rather than by pointing cargo at a
# per-engine Cargo.toml that no longer exists.
add_test(NAME spectro_core
         COMMAND ${SPECTRO_ENV} ${SPECTRO_CARGO} test --manifest-path
                 ${CMAKE_SOURCE_DIR}/Cargo.toml -p spectro-core -p spectro-capi)
